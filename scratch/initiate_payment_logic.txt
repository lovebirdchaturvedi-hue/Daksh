<script>
    let currentCurrency = 'INR';
    let selectedPlan = { name: '', usd: 0, inr: 0 };

    function toggleCurrency() {
        currentCurrency = currentCurrency === 'USD' ? 'INR' : 'USD';
        const knob = document.getElementById('toggle-knob');
        const lUSD = document.getElementById('label-usd');
        const lINR = document.getElementById('label-inr');
        
        document.body.classList.remove('currency-inr', 'currency-usd');
        document.body.classList.add(currentCurrency === 'INR' ? 'currency-inr' : 'currency-usd');

        if (currentCurrency === 'INR') {
            knob.style.left = '32px';
            lINR.style.opacity = '1';
            lINR.style.fontWeight = '700';
            lINR.style.color = 'var(--gold)';
            lUSD.style.opacity = '0.6';
            lUSD.style.fontWeight = '400';
            lUSD.style.color = '#fff';
        } else {
            knob.style.left = '2px';
            lUSD.style.opacity = '1';
            lUSD.style.fontWeight = '700';
            lUSD.style.color = 'var(--gold)';
            lINR.style.opacity = '0.6';
            lINR.style.fontWeight = '400';
            lINR.style.color = '#fff';
        }
    }

    function initiatePayment(planName, usd, inr) {
        selectedPlan = { name: planName, usd, inr };
        
        let usdEquiv = 0;
        if (inr === 59000) usdEquiv = 499;
        else if (inr === 168000) usdEquiv = 2000;
        else if (inr === 294000) usdEquiv = 3500;
        else usdEquiv = usd;

        document.getElementById('modalPlanName').innerText = planName;
        document.getElementById('modalPriceUSD').innerText = '$' + usd.toLocaleString();
        document.getElementById('modalPriceINR').innerText = '₹' + inr.toLocaleString();
        document.getElementById('modalPriceINRManual').innerText = '₹' + inr.toLocaleString();
        document.getElementById('modalPriceUSDEquiv').innerText = '$' + usdEquiv.toLocaleString() + '.00';
        
        document.getElementById("paymentModal").style.display = "flex";
        
        if (currentCurrency === 'INR') { switchPayment('qr'); } else { switchPayment('paypal'); }
        
        renderPayPal(usd, planName, "paypalButtons");
        if (usdEquiv > 0) { renderPayPal(usdEquiv, planName + " (INR Base)", "paypalButtonsINR"); }
    }

    function switchPayment(method) {
        const tPayPal = document.getElementById('tab-paypal');
        const tQR = document.getElementById('tab-qr');
        const tManual = document.getElementById('tab-manual');
        const sPayPal = document.getElementById('paypalSection');
        const sQR = document.getElementById('qrSection');
        const sManual = document.getElementById('manualSection');
        
        tPayPal.style.background = 'transparent'; tPayPal.style.color = '#94a3b8';
        tQR.style.background = 'transparent'; tQR.style.color = '#94a3b8';
        tManual.style.background = 'transparent'; tManual.style.color = '#94a3b8';
        sPayPal.style.display = 'none';
        sQR.style.display = 'none';
        sManual.style.display = 'none';

        if (method === 'paypal') {
            tPayPal.style.background = 'var(--gold)'; tPayPal.style.color = 'black';
            sPayPal.style.display = 'block';
        } else if (method === 'qr') {
            tQR.style.background = 'var(--gold)'; tQR.style.color = 'black';
            sQR.style.display = 'block';
        } else {
            tManual.style.background = 'var(--gold)'; tManual.style.color = 'black';
            sManual.style.display = 'block';
        }
    }

    function renderPayPal(amount, planName, containerId) {
        const container = document.getElementById(containerId);
        if (!container) return;
        container.innerHTML = "";
        setTimeout(function () {
            paypal.Buttons({
                createOrder: function (data, actions) {
                    return actions.order.create({
                        purchase_units: [{
                            description: planName + " – APD Global Trade",
                            amount: { currency_code: "USD", value: amount.toFixed(2) }
                        }]
                    });
                },
                onApprove: function (data, actions) {
                    return actions.order.capture().then(function () {
                        window.location.href = "/payment-status.html";
                    });
                }
            }).render("#" + containerId);
        }, 400);
    }

    async function initiatePhonePe() {
        const btn = document.getElementById('phonepeBtn');
        const err = document.getElementById('phonepeError');
        btn.innerText = 'Initializing...';
        btn.disabled = true;
        try {
            const res = await fetch('/api/phonepe-init', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ amount: selectedPlan.inr, planName: selectedPlan.name })
            });
            const data = await res.json();
            if (data.success && data.url) { window.location.href = data.url; } 
            else { err.innerText = data.error || 'Failed.'; err.style.display = 'block'; btn.innerText = 'PAY VIA PHONEPE'; btn.disabled = false; }
        } catch(e) { err.innerText = 'Network error.'; err.style.display = 'block'; btn.innerText = 'PAY VIA PHONEPE'; btn.disabled = false; }
    }

    function closeModal() { document.getElementById("paymentModal").style.display = "none"; }
    function toggleDrawer() {
        document.getElementById('sideDrawer').classList.toggle('active');
        document.getElementById('drawerOverlay').classList.toggle('active');
    }
    
    // Force render on load
    setTimeout(() => { currentCurrency = 'USD'; toggleCurrency(); }, 100);
  </script>

<script>
    function toggleWaHub() {
        const hub = document.getElementById('waHub');
        hub.classList.toggle('active');
    }

    // Auto-detect GeoIP and switch to USD if outside India
    document.addEventListener("DOMContentLoaded", () => {
        fetch('https://ipapi.co/json/')
            .then(res => res.json())
            .then(data => {
                if (data.country_code && data.country_code !== 'IN') {
                    if (typeof currentCurrency !== 'undefined' && currentCurrency === 'INR' && typeof toggleCurrency === 'function') {
                        // Switch knob and prices to USD
                        toggleCurrency();
                    }
                }
            })
            .catch(err => console.log('GeoIP detection failed'));

        // Exit Intent Popup Logic
        let hasShownExitIntent = false;

        // Desktop Exit Intent
        document.addEventListener('mouseleave', (e) => {
            if (e.clientY < 0 && !hasShownExitIntent) {
                showExitIntent();
            }
        });

        // Mobile Exit Intent (Fast scroll up)
        let lastScrollTop = window.scrollY || document.documentElement.scrollTop;
        let lastScrollTime = Date.now();
        
        window.addEventListener('scroll', () => {
            if (hasShownExitIntent) return;
            let currentScrollTop = window.scrollY || document.documentElement.scrollTop;
            let currentTime = Date.now();
            let timeDiff = currentTime - lastScrollTime;
            let scrollDiff = lastScrollTop - currentScrollTop;

            // If scrolling UP fast (> 100px in < 200ms)
            if (scrollDiff > 100 && timeDiff < 200) {
                showExitIntent();
            }
            
            lastScrollTop = currentScrollTop;
            lastScrollTime = currentTime;
        }, { passive: true });

        // Timer Fallback (Show after 60s if not shown)
        setTimeout(() => {
            if (!hasShownExitIntent) {
                showExitIntent();
            }
        }, 60000);

        function showExitIntent() {
            // Check session storage to not annoy users
            if (sessionStorage.getItem('exitIntentShown')) return;
            
            document.getElementById('exitIntentModal').style.display = 'flex';
            hasShownExitIntent = true;
            sessionStorage.setItem('exitIntentShown', 'true');
        }
    });

    window.addEventListener('DOMContentLoaded', () => {
        const urlParams = new URLSearchParams(window.location.search);
        if (urlParams.get('buy') === '3') {
            setTimeout(() => {
                initiatePayment('3 Verified Buyers', 59, 4999);
            }, 500);
        }
    });
  </script>
